<script>

  /***********************

   * CONFIG — CHANGE THESE

  const SUPABASE_PROJECT_URL = "https://pzgnggkanppnwnkjgwrm.supabase.co";

  const EDGE_SCAN_READY_PATH = "/functions/v1/warehouse_scan_ready";

  const EDGE_COUNTS_PATH     = "/functions/v1/warehouse_ready_counts";


  // ✅ Put your current working token here for now.

  // Later we replace with proper login and session token.

  const AUTH_BEARER_TOKEN = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6InB6Z25nZ2thbnBwbndua2pnd3JtIiwicm9sZSI6ImFub24iLCJpYXQiOjE3NjkzMjAxODEsImV4cCI6MjA4NDg5NjE4MX0.YfKYz3XlnX3U8Nykq7CglD8lHpgnUq-Azk2r-xorX5k";


  // Optional but safe in Edge calls:

  const SUPABASE_ANON_KEY = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6InB6Z25nZ2thbnBwbndua2pnd3JtIiwicm9sZSI6ImFub24iLCJpYXQiOjE3NjkzMjAxODEsImV4cCI6MjA4NDg5NjE4MX0.YfKYz3XlnX3U8Nykq7CglD8lHpgnUq-Azk2r-xorX5k";


  // Temporary operator name (later: from login)

  const OPERATOR_NAME = "WAREHOUSE_USER";


  /***********************

   * STATE

   ***********************/

  let batchCount = 0;

  let todayReadyCount = null;


  const accepted = []; // [{awb, ts}]

  const MAX_ACCEPTED = 30;


  // If an error modal is open, pause camera-driven scans

  let errorBlocking = false;


  // Allow multiple in-flight calls? We keep it lightweight:

  // If operator scans too fast, we still send calls.

  // But if error happens, we block until OK.

  let inFlight = 0;


  const $ = (id) => document.getElementById(id);


  function setStatus(dotClass, text, sub) {

    const dot = $("statusDot");

    dot.className = "dot" + (dotClass ? " " + dotClass : "");

    $("statusText").textContent = text;

    $("statusSub").textContent = sub || "";

  }


  function showErrorPopup(message) {

    errorBlocking = true;

    $("modalMsg").textContent = message;

    $("modalBackdrop").style.display = "flex";

    setStatus("red", "REJECTED", "Fix parcel and press OK");

  }


  function hideErrorPopup() {

    $("modalBackdrop").style.display = "none";

    errorBlocking = false;

    $("awbInput").focus();

    setStatus("", "Ready to scan", `In-flight: ${inFlight}`);

  }


  function renderAcceptedList() {

    const ul = $("acceptedList");

    ul.innerHTML = "";

    for (const item of accepted) {

      const li = document.createElement("li");

      li.className = "li";

      li.innerHTML = `

        <div>

          <div class="awb">${item.awb} <span style="margin-left:8px; display:inline-block; width:8px; height:8px; border-radius:50%; background:#37d67a;"></span></div>

          <div class="small">${new Date(item.ts).toLocaleTimeString()}</div>

        </div>

      `;

      ul.appendChild(li);

    }

  }


  function setCounters() {

    $("batchCount").textContent = String(batchCount);

    $("todayReady").textContent = (todayReadyCount === null) ? "—" : String(todayReadyCount);

  }


  function todayIST() {

    const now = new Date();

    const utc = now.getTime() + (now.getTimezoneOffset() * 60000);

    const ist = new Date(utc + (5.5 * 3600000));

    const yyyy = ist.getFullYear();

    const mm = String(ist.getMonth() + 1).padStart(2, "0");

    const dd = String(ist.getDate()).padStart(2, "0");

    return `${yyyy}-${mm}-${dd}`;

  }


  async function fetchTodayCount() {

    try {

      const date = todayIST();

      const url = `${SUPABASE_PROJECT_URL}${EDGE_COUNTS_PATH}?date=${encodeURIComponent(date)}`;

      const res = await fetch(url, {

        method: "GET",

        headers: {

          "Authorization": `Bearer ${AUTH_BEARER_TOKEN}`,

          "apikey": SUPABASE_ANON_KEY

        }

      });

      const json = await res.json();

      if (!json.ok) throw new Error(json.message || "counts failed");

      todayReadyCount = json.today_ready_count || 0;

      setCounters();

    } catch (e) {

      console.log("counts_error", e);

      todayReadyCount = null;

      setCounters();

    }

  }


  // SUCCESS should not interrupt; ERROR should interrupt (popup)

  async function submitAwb(awb) {

    const clean = String(awb || "").trim();

    if (!clean) return;


    if (errorBlocking) return; // don't accept new scans while error popup open


    inFlight += 1;

    setStatus("yellow", "Processing…", `${clean} | In-flight: ${inFlight}`);


    try {

      const url = `${SUPABASE_PROJECT_URL}${EDGE_SCAN_READY_PATH}`;

      const res = await fetch(url, {

        method: "POST",

        headers: {

          "Content-Type": "application/json",

          "Authorization": `Bearer ${AUTH_BEARER_TOKEN}`,

          "apikey": SUPABASE_ANON_KEY

        },

        body: JSON.stringify({ awb: clean, operator: OPERATOR_NAME })

      });


      const json = await res.json();


      if (json.result === "ACCEPTED") {

        batchCount += 1;

        if (typeof todayReadyCount === "number") todayReadyCount += 1;


        accepted.unshift({ awb: clean, ts: Date.now() });

        if (accepted.length > MAX_ACCEPTED) accepted.pop();


        setCounters();

        renderAcceptedList();


        // tiny success signal, no blocking

        setStatus("green", "ACCEPTED", `${clean} | In-flight: ${inFlight}`);

        setTimeout(() => {

          if (!errorBlocking) setStatus("", "Ready to scan", `In-flight: ${inFlight}`);

        }, 250);


      } else {

        const msg = json.message || "Rejected";

        showErrorPopup(`${clean}\n\n${msg}`);

      }


    } catch (e) {

      showErrorPopup(`${clean}\n\nServer error: ${e?.message || e}`);

    } finally {

      inFlight = Math.max(0, inFlight - 1);

      $("awbInput").value = "";

      $("awbInput").focus();

      if (!errorBlocking) setStatus("", "Ready to scan", `In-flight: ${inFlight}`);

    }

  }


  // Input handlers

  $("submitBtn").addEventListener("click", () => submitAwb($("awbInput").value));

  $("awbInput").addEventListener("keydown", (e) => {

    if (e.key === "Enter") submitAwb($("awbInput").value);

  });


  $("resetBatch").addEventListener("click", () => {

    batchCount = 0;

    accepted.length = 0;

    renderAcceptedList();

    setCounters();

    setStatus("", "Batch reset", "Ready…");

    setTimeout(() => {

      if (!errorBlocking) setStatus("", "Ready to scan", `In-flight: ${inFlight}`);

    }, 350);

  });


  $("closeModal").addEventListener("click", hideErrorPopup);

  $("modalBackdrop").addEventListener("click", (e) => {

    if (e.target === $("modalBackdrop")) hideErrorPopup();

  });


  /***********************

   * CAMERA SCANNING

   ***********************/

  let stream = null;

  let scanning = false;

  let detector = null;


  async function startCamera() {

    if (scanning) return;

    if (!("BarcodeDetector" in window)) {

      showErrorPopup("This device/browser doesn't support native BarcodeDetector.\nUse input box or we can add QuaggaJS fallback later.");

      return;

    }

    detector = new BarcodeDetector({ formats: ["code_128", "ean_13", "ean_8", "qr_code"] });


    stream = await navigator.mediaDevices.getUserMedia({

      video: { facingMode: "environment" },

      audio: false

    });


    const video = $("video");

    video.srcObject = stream;

    await video.play();


    scanning = true;

    $("toggleCamera").textContent = "Stop camera";

    requestAnimationFrame(scanLoop);

  }


  function stopCamera() {

    scanning = false;

    $("toggleCamera").textContent = "Start camera";

    const video = $("video");

    if (video) video.pause();

    if (stream) {

      stream.getTracks().forEach(t => t.stop());

      stream = null;

    }

  }


  // Frame-level guard: prevents same barcode firing continuously from camera frames

  let lastSeen = "";

  let lastSeenAt = 0;


  async function scanLoop() {

    if (!scanning) return;


    const video = $("video");

    try {

      if (!errorBlocking) {

        const barcodes = await detector.detect(video);

        if (barcodes && barcodes.length) {

          const raw = (barcodes[0].rawValue || "").trim();

          const now = Date.now();


          // Prevent rapid repeats from same frame stream (not a 2-3 sec lock)

          if (raw && (raw !== lastSeen || (now - lastSeenAt) > 800)) {

            lastSeen = raw;

            lastSeenAt = now;

            submitAwb(raw);

          }

        }

      }

    } catch (_) {}


    requestAnimationFrame(scanLoop);

  }


  $("toggleCamera").addEventListener("click", async () => {

    try {

      if (scanning) stopCamera();

      else await startCamera();

    } catch (e) {

      stopCamera();

      showErrorPopup("Camera start failed.\n" + (e?.message || e));

    }

  });


  // Init

  setCounters();

  setStatus("", "Ready to scan", "Waiting…");

  $("awbInput").focus();

  fetchTodayCount();

</script>
